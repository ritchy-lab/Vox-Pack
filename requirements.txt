# Effet robot (traitement du WAV).
# Deux versions, pour une raison vue sur le PC de Richard le 3 novembre 2026 :
# numpy 2.4.6 s'installe mais ne se charge pas sur un Windows 10 ancien
# (1903, « DLL initialization routine failed »), alors que 1.26.4 y passe.
# macOS garde la 2.4.6, qui n'a jamais posé problème.
numpy==2.4.6; sys_platform == "darwin"
numpy==1.26.4; sys_platform != "darwin"

# macOS uniquement : sélecteur de dossier natif (feuille NSOpenPanel)
# en français, à la place de celui de Tk qui écrit « Choose » en dur.
pyobjc-framework-Cocoa==12.2.2; sys_platform == "darwin"

# --- Uniquement en dehors de macOS ---
# Sur macOS, la conversion audio utilise afconvert et la lecture afplay, tous
# deux fournis par le système. Ailleurs (Windows), tout se fait en Python :
#   soundfile  lit le MP3 (via libsndfile)
#   soxr       le rééchantillonne en 32 kHz
#   sounddevice joue l'aperçu (via PortAudio)
# numpy vient avec soundfile.
soundfile==0.14.0
soxr==1.1.0
sounddevice==0.5.6

# --- Uniquement pour fabriquer l'application (./build.sh) ---
